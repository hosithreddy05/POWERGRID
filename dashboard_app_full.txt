    11	# ============================================================
    12	# HTML / MARKDOWN RENDERING HELPER
    13	# ============================================================
    14	
    15	def render_markdown(content, **kwargs):
    16	    """
    17	    Render HTML safely without allowing Python indentation
    18	    to turn HTML into a Markdown code block.
    19	    """
    20	
    21	    if not isinstance(content, str):
    22	        return st.markdown(content, **kwargs)
    23	
    24	    cleaned = content.strip()
    25	
    26	    # HTML/CSS blocks should be rendered as HTML directly.
    27	    if (
    28	        cleaned.startswith("<")
    29	        or "<div" in cleaned
    30	        or "<style" in cleaned
    31	        or "</div>" in cleaned
    32	    ):
    33	        try:
    34	            st.html(cleaned)
    35	            return
    36	        except AttributeError:
    37	            # Fallback for older Streamlit versions.
    38	            lines = cleaned.splitlines()
    39	            cleaned = "\n".join(line.lstrip() for line in lines)
    40	
    41	    return st.markdown(
    42	        cleaned,
    43	        **kwargs
    44	    )
